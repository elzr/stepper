## Generator architecture (Lua)

```lua
-- features/L008-keymap/keymap.lua
local M = {}
local projectRoot, debounceTimer = nil, nil
M._watchers = {}  -- module-scope to avoid GC (per L005 lesson)

function M.generate()
  local rcmd       = readRcmdPlist()
  local hyper      = readHyperConfigs()  -- bear-notes + live-toggle + hyper-actions
  local annot      = readAnnotationsJsonc()
  local warnings   = computeDrift(rcmd, hyper, annot)
  local html       = renderHtml(rcmd, hyper, annot, warnings)
  writeFile(projectRoot .. "features/L008-keymap/keymap.html", html)
  print(string.format("[keymap] Regenerated (%d warnings)", #warnings))
end

local function debounce()
  if debounceTimer then debounceTimer:stop() end
  debounceTimer = hs.timer.doAfter(0.5, M.generate)  -- coalesces editor save bursts
end

function M.init(root)
  projectRoot = root
  M.generate()
  for _, p in ipairs({
    rcmdPlistPath(),
    root .. "data/live-toggle-hotkeys.json",
    root .. "data/bear-notes.jsonc",
    root .. "data/hyper-actions.jsonc",
    root .. "features/L008-keymap/annotations.jsonc",
  }) do
    table.insert(M._watchers, hs.pathwatcher.new(p, debounce):start())
  end
end

return M
```

Wired in `lua/stepper.lua` near other module loads. L005's weekly trigger requires no change — it reloads HS, which re-runs `M.init()` → `M.generate()`.
